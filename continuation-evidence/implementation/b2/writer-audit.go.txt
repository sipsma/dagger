package main
import("bytes";"fmt";"go/ast";"go/format";"go/parser";"go/token";"io/fs";"os";"path/filepath";"strings")
func main(){fset:=token.NewFileSet();for _,root:=range []string{"core","engine/server"}{filepath.WalkDir(root,func(path string,entry fs.DirEntry,err error)error{if err!=nil{return err};if entry.IsDir(){if entry.Name()=="testdata" {return filepath.SkipDir};return nil};if !strings.HasSuffix(path,".go")||strings.HasSuffix(path,"_test.go"){return nil};f,err:=parser.ParseFile(fset,path,nil,0);if err!=nil{panic(err)};for _,decl:=range f.Decls{fn,ok:=decl.(*ast.FuncDecl);if !ok {continue};name:=fn.Name.Name;if fn.Recv!=nil {var b bytes.Buffer;format.Node(&b,fset,fn.Recv.List[0].Type);name=b.String()+"."+name};emit:=func(n ast.Node,kind string){var b bytes.Buffer;format.Node(&b,fset,n);fmt.Printf("%s:%d | %s | %s | %s\n",path,fset.Position(n.Pos()).Line,name,kind,strings.ReplaceAll(b.String(),"\n"," "))};ast.Inspect(fn.Body,func(n ast.Node)bool{switch n:=n.(type){case *ast.CallExpr:if sel,ok:=n.Fun.(*ast.SelectorExpr);ok {switch sel.Sel.Name{case "SetPath","SetSnapshot","clearLazy","finishLazyLocked":emit(n,"publication")}};case *ast.CompositeLit:typ:="";switch t:=n.Type.(type){case *ast.Ident:typ=t.Name;case *ast.SelectorExpr:typ=t.Sel.Name};if typ=="File"||typ=="Directory" {for _,elem:=range n.Elts{kv,ok:=elem.(*ast.KeyValueExpr);if !ok {continue};key,ok:=kv.Key.(*ast.Ident);if !ok{continue};switch key.Name{case "File","Dir","Snapshot","Lazy","completedRecipe","completedRecipeJSON","completedRecipeKind":emit(kv,"private initializer "+typ)}}};case *ast.AssignStmt:if path=="core/file.go"||path=="core/directory.go"||path=="core/filesystem_output.go"{for _,lhs:=range n.Lhs{if sel,ok:=lhs.(*ast.SelectorExpr);ok{switch sel.Sel.Name{case "completedRecipe","completedRecipeKind","completedRecipeJSON","Lazy":emit(n,"producer assignment")}}}}};return true})};return nil})};_ = os.Stdout}
