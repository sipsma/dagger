package schema

import (
	"context"
	"encoding/json"
	"path/filepath"
	"testing"

	"github.com/dagger/dagger/dagql"
	"github.com/dagger/dagger/dagql/call"
	"github.com/dagger/dagger/engine/snapshots/config"
	"github.com/stretchr/testify/require"
)

func TestB4EagerContainerProducerProbe(t *testing.T) {
	env := &persistedSchemaTestEnv{dbPath: filepath.Join(t.TempDir(), "cache.db")}
	ctx, cache, srv := env.open(t)
	data, err := srv.Query(ctx, `{container(platform:"linux/amd64"){withWorkdir(path:"/src"){id}}}`, nil)
	require.NoError(t, err)
	raw, err := json.Marshal(data)
	require.NoError(t, err)
	var out struct {
		Container struct{ WithWorkdir struct{ ID string } }
	}
	require.NoError(t, json.Unmarshal(raw, &out))
	id := new(call.ID)
	require.NoError(t, id.Decode(out.Container.WithWorkdir.ID))
	res, err := cache.LoadResultByResultID(ctx, "", srv, id.EngineResultID())
	require.NoError(t, err)
	record, err := cache.CapturePersistedRecord(ctx, res)
	require.NoError(t, err)
	var payload struct {
		LazyJSON json.RawMessage `json:"lazyJSON"`
	}
	require.NoError(t, json.Unmarshal(record.Envelope.ObjectJSON, &payload))
	t.Logf("field=%s savedRecipe=%s payload=%s", record.Call.Field, payload.LazyJSON, record.Envelope.ObjectJSON)
	require.Equal(t, "withWorkdir", record.Call.Field)
	require.Empty(t, payload.LazyJSON, "the completed eager schema branch stores no saved producer")
	require.NoError(t, cache.WithExportedValues(ctx, dagql.ValueSelection{Roots: []dagql.AnyResult{res}}, config.RefConfig{}, func(_ context.Context, values *dagql.ExportedValues) error {
		for _, v := range values.Bundle.Values {
			if v.Record.Call.Field == "withWorkdir" {
				t.Logf("foreign payload=%s", v.Record.Envelope.ObjectJSON)
			}
		}
		return nil
	}))
}
